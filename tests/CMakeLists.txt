# DragonGL - standalone unit-test project (tests/)
#
# This directory is SELF-CONTAINED: the main project's CMakeLists.txt has
# no reference to it anymore. Build and run the suite independently:
#
#   cmake -B build-tests tests
#   cmake --build build-tests
#   ctest --test-dir build-tests --output-on-failure
#
# The test sources live here; the code under test is pulled straight from
# ../src (rules.c, pathfinding.c, client_tome.c) and the headers from
# ../include, so the tests always exercise the real production sources.
# Compile settings mirror the main project (clang, C11, same warning
# flags) so a warning can never hide behind different flags.
# =========================================================================

cmake_minimum_required(VERSION 3.16)
# Same compiler policy as the main CMakeLists.txt: clang by default, but
# never overriding an explicit choice (-DCMAKE_C_COMPILER or $CC).
option(DRAGONGL_USE_CLANG "Default to clang as the C compiler (overridable)" ON)
if(DRAGONGL_USE_CLANG AND NOT DEFINED CMAKE_C_COMPILER AND NOT DEFINED ENV{CC})
    set(CMAKE_C_COMPILER clang)
endif()
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/DragonGL.cmake")
project(DragonGLTests VERSION ${DRAGONGL_VERSION} LANGUAGES C)

# Same flags as the main project (cmake/DragonGL.cmake).
dragongl_compile_settings()

# Project roots (this file is tests/CMakeLists.txt)
set(DG_ROOT   "${CMAKE_CURRENT_SOURCE_DIR}/..")
set(DG_SRCS   "${DG_ROOT}/src")
set(DG_INC    "${DG_ROOT}/include")

# Include paths: the shared headers (include/) plus the client and server
# source dirs, where client_tome.h / pathfinding.h live.
include_directories(${DG_INC} ${DG_SRCS}/client ${DG_SRCS}/server)

enable_testing()
find_package(Threads REQUIRED)

# --- rules engine (dice, modifiers, damage, conditions) ---
add_executable(test_rules test_rules.c ${DG_SRCS}/shared/rules.c)
target_link_libraries(test_rules m)
add_test(NAME rules COMMAND test_rules)

# --- spell effects: the parser of data/spells.json "effect" and the
# --- resolution of an effect against one target (forced d20)
add_executable(test_spell_rules test_spell_rules.c ${DG_SRCS}/shared/spell_rules.c
    ${DG_SRCS}/shared/spell_effect.c ${DG_SRCS}/shared/rules.c ${DRAGONGL_CJSON_SRCS})
target_link_libraries(test_spell_rules m ${DRAGONGL_CJSON_LIBS})
add_test(NAME spell_rules COMMAND test_spell_rules)

# --- spell balance: every damage spell of data/spells.json simulated with
# --- the game's loader and resolution, against tests/spell_balance.json
add_executable(spell_balance spell_balance.c ${DG_SRCS}/shared/data_loader.c
    ${DG_SRCS}/shared/data_items.c ${DG_SRCS}/shared/data_item_effects.c
    ${DG_SRCS}/shared/spell_rules.c ${DG_SRCS}/shared/spell_effect.c ${DG_SRCS}/shared/rules.c
    ${DRAGONGL_CJSON_SRCS} ${DG_SRCS}/shared/strnum.c ${DG_SRCS}/shared/binio.c)
target_link_libraries(spell_balance m ${DRAGONGL_CJSON_LIBS})
add_test(NAME spell_balance
         COMMAND spell_balance ${DG_ROOT}/data ${CMAKE_CURRENT_SOURCE_DIR}/spell_balance.json)

# --- A* pathfinding (fixed hand-built maps) ---
add_executable(test_pathfinding test_pathfinding.c ${DG_SRCS}/server/pathfinding.c)
target_link_libraries(test_pathfinding m)
add_test(NAME pathfinding COMMAND test_pathfinding)

# --- wire protocol: encode/decode round trips and malformed payloads ---
add_executable(test_protocol test_protocol.c ${DG_SRCS}/shared/wire.c ${DG_SRCS}/shared/binio.c)
add_test(NAME protocol COMMAND test_protocol)

# --- wire protocol: conformance with protocol/messages.json (generated by
# --- tools/gen_protocol.py: every example and every invalid variant)
add_executable(test_protocol_conformance test_protocol_conformance.c
    ${DG_SRCS}/shared/wire.c ${DG_SRCS}/shared/binio.c)
target_link_libraries(test_protocol_conformance m)
add_test(NAME protocol_conformance COMMAND test_protocol_conformance)

# --- TLS helpers and the known_servers store (trust on first use) ---
find_package(OpenSSL 3.0 REQUIRED)
add_executable(test_tls test_tls.c ${DG_SRCS}/shared/tls_common.c
    ${DG_SRCS}/client/known_servers.c ${DG_SRCS}/shared/net.c ${DG_SRCS}/shared/strnum.c)
target_link_libraries(test_tls OpenSSL::SSL OpenSSL::Crypto)
add_test(NAME tls COMMAND test_tls)

# (DragonGL+: the floating-tome test exercises the upstream client's
# client_tome.c; the DragonGL+ client keeps its own.)

# --- DragonGL+: codec of the protocol extension (protocol_plus.h) ---
add_executable(test_protocol_plus test_protocol_plus.c ${DG_SRCS}/shared/wire_plus.c
    ${DG_SRCS}/shared/wire.c ${DG_SRCS}/shared/binio.c)
add_test(NAME protocol_plus COMMAND test_protocol_plus)

# (DragonGL+: the tests of the upstream Vulkan client -- command line,
# HUD theme, VFX pools and GPU mesh generation -- are not built: the
# DragonGL+ client has its own UI, effects and renderers.)
